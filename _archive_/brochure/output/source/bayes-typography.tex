% Project-local fonts make the brochure deterministic and preserve the brand
% type system even when Manrope or Newsreader are not installed system-wide.
\IfFileExists{assets/fonts/manrope/Manrope-Regular.ttf}{
  \setsansfont{Manrope-Regular.ttf}[
    Path=assets/fonts/manrope/,
    BoldFont=Manrope-ExtraBold.ttf,
    ItalicFont=Manrope-Regular.ttf,
    BoldItalicFont=Manrope-ExtraBold.ttf,
    Ligatures=TeX
  ]
  \newfontfamily\manrope{Manrope-Regular.ttf}[
    Path=assets/fonts/manrope/,
    BoldFont=Manrope-ExtraBold.ttf,
    Ligatures=TeX
  ]
}{
  \IfFontExistsTF{Manrope}{
    \setsansfont{Manrope}[Ligatures=TeX]
    \newfontfamily\manrope{Manrope}[Ligatures=TeX]
  }{
    \setsansfont{TeX Gyre Heros}[Ligatures=TeX]
    \newfontfamily\manrope{TeX Gyre Heros}[Ligatures=TeX]
  }
}

\IfFileExists{assets/fonts/newsreader/Newsreader-Variable.ttf}{
  \newfontfamily\newsreader{Newsreader-Variable.ttf}[
    Path=assets/fonts/newsreader/,
    ItalicFont=Newsreader-Italic-Variable.ttf,
    Ligatures=TeX
  ]
}{
  \IfFontExistsTF{Newsreader}{
    \newfontfamily\newsreader{Newsreader}[Ligatures=TeX]
  }{
    \newfontfamily\newsreader{TeX Gyre Schola}[Ligatures=TeX]
  }
}

\IfFileExists{assets/fonts/caveat/Caveat-VariableFont_wght.ttf}{
  \newfontfamily\caveat{Caveat-VariableFont_wght.ttf}[
    Path=assets/fonts/caveat/,
    Ligatures=TeX
  ]
}{
  \IfFontExistsTF{Caveat}{
    \newfontfamily\caveat{Caveat}[Ligatures=TeX]
  }{
    \newfontfamily\caveat{TeX Gyre Schola}[Ligatures=TeX]
  }
}
